-- DI CHUYỂN AN TOÀN (SafeMove từ HDanh, giữ nguyên)

local STEP_SIZE = 60
local STEP_WAIT = 0.5

function SafeMove(targetCF, checkStop)
    local hrp = Character:FindFirstChild("HumanoidRootPart")
    if not hrp then return end

    local targetPos = targetCF.Position
    while true do
        if checkStop and checkStop() then break end
        local curPos    = hrp.Position
        local remaining = (targetPos - curPos).Magnitude
        if remaining < 8 then break end

        local dir      = (targetPos - curPos).Unit
        local stepDist = math.min(remaining, STEP_SIZE)
        local nextCF   = (remaining <= STEP_SIZE) and targetCF
                         or CFrame.new(curPos + dir * stepDist)

        pcall(function() hrp.CFrame = nextCF end)
        task.wait(STEP_WAIT)
    end
end

local _tweenThread = nil

function Tween(targetCF)
    if not targetCF then return end
    if _tweenThread then
        pcall(function() task.cancel(_tweenThread) end)
        _tweenThread = nil
    end
    _tweenThread = task.spawn(function()
        SafeMove(targetCF, function() return _tweenThread == nil end)
        _tweenThread = nil
    end)
end

function TweenBlock(targetCF)
    if not targetCF then return end
    if _tweenThread then
        pcall(function() task.cancel(_tweenThread) end)
        _tweenThread = nil
    end
    _G.Clip2 = true
    pcall(function()
        SafeMove(targetCF, function() return _G.StopTween end)
    end)
    _G.Clip2 = false
end

function AttackNoCoolDown()
    local tool = Character:FindFirstChildOfClass("Tool")
    if not tool then return end

    local enemies = Workspace:FindFirstChild("Enemies")
    if not enemies then return end

    if tool:FindFirstChild("LeftClickRemote") then
        for _, mob in pairs(enemies:GetChildren()) do
            local hrp = mob:FindFirstChild("HumanoidRootPart")
            local hum = mob:FindFirstChild("Humanoid")
            if hrp and hum and hum.Health > 0 then
                local dir = (hrp.Position - Root.Position).Unit
                pcall(function() tool.LeftClickRemote:FireServer(dir, 1) end)
            end
        end
    else
        local net    = ReplicatedStorage:FindFirstChild("Modules")
        local modNet = net and net:FindFirstChild("Net")
        if not modNet then return end
        local regAtk = modNet:FindFirstChild("RE/RegisterAttack")
        local regHit = modNet:FindFirstChild("RE/RegisterHit")
        if not regAtk or not regHit then return end

        local hits = {}
        local head
        for _, mob in pairs(enemies:GetChildren()) do
            local hum = mob:FindFirstChild("Humanoid")
            local hrp = mob:FindFirstChild("HumanoidRootPart")
            if hum and hrp and hum.Health > 0
               and (hrp.Position - Root.Position).Magnitude <= 60 then
                local h = mob:FindFirstChild("Head")
                if h then
                    hits[#hits + 1] = {mob, h}
                    head = head or h
                end
            end
        end
        if #hits > 0 then
            pcall(function()
                regAtk:FireServer(1e-9)
                regHit:FireServer(head, hits)
            end)
        end
    end
end

-- TABS (giữ wrapper từ HDanh — chạy được ổn)
-- ============================================================
local function makeProxy(obj, holder)
    local proxy = {}
    setmetatable(proxy, {
        __index = function(_, k)
            if k == "OnChanged" then
                return function(_, fn) holder.extra = fn return proxy end
            end
            if k == "SetValue" then
                return function(_, v)
                    if obj.SetValue then pcall(obj.SetValue, v) end
                end
            end
            if k == "SetDesc" or k == "SetText" then
                return function(_, t)
                    if obj.SetText then pcall(obj.SetText, obj, t) end
                end
            end
            local v = rawget(obj, k) or (type(obj) == "table" and obj[k])
            if type(v) == "function" then
                return function(_, ...) return pcall(v, obj, ...) end
            end
            return v
        end
    })
    return proxy
end

local function wrapTab(rawTab)
    local _currentSection = nil
    local _nextIsRight = false

    local function ensureSection()
        if not _currentSection then
            _currentSection = rawTab:AddLeftGroupbox(" ")
        end
    end

    local wrapped = {}

    function wrapped:AddSection(name)
        if _nextIsRight then
            _currentSection = rawTab:AddRightGroupbox(name or " ")
            _nextIsRight = false
        else
            _currentSection = rawTab:AddLeftGroupbox(name or " ")
            _nextIsRight = true
        end
        return _currentSection
    end

    function wrapped:AddToggle(id, setting)
        ensureSection()
        local holder = { extra = nil }
        local origCb = setting.Callback
        setting.Callback = function(v)
            if origCb then pcall(origCb, v) end
            if holder.extra then pcall(holder.extra, v) end
        end
        setting.Description = nil
        return makeProxy(_currentSection:AddToggle(id, setting), holder)
    end

    function wrapped:AddButton(setting, cb)
        ensureSection()
        if type(setting) == "table" then setting.Description = nil end
        return makeProxy(_currentSection:AddButton(setting, cb), {})
    end

    function wrapped:AddDropdown(id, setting)
        ensureSection()
        local holder = { extra = nil }
        local origCb = setting.Callback
        setting.Callback = function(v)
            if origCb then pcall(origCb, v) end
            if holder.extra then pcall(holder.extra, v) end
        end
        setting.Description = nil
        return makeProxy(_currentSection:AddDropdown(id, setting), holder)
    end

    function wrapped:AddSlider(id, setting)
        ensureSection()
        local holder = { extra = nil }
        local origCb = setting.Callback
        setting.Callback = function(v)
            if origCb then pcall(origCb, v) end
            if holder.extra then pcall(holder.extra, v) end
        end
        return makeProxy(_currentSection:AddSlider(setting), holder)
    end

    function wrapped:AddInput(id, setting)
        ensureSection()
        local holder = { extra = nil }
        local origCb = setting.Callback
        setting.Callback = function(v)
            if origCb then pcall(origCb, v) end
            if holder.extra then pcall(holder.extra, v) end
        end
        return makeProxy(_currentSection:AddInput(id, setting), holder)
    end

    function wrapped:AddParagraph(setting)
        ensureSection()
        local txt = (setting.Title or "") .. "\n" .. (setting.Description or setting.Desc or "")
        return makeProxy(_currentSection:AddLabel(txt), {})
    end

    function wrapped:AddLabel(text)
        ensureSection()
        return makeProxy(_currentSection:AddLabel(text), {})
    end

    return wrapped
end

ESPPlayer = false

function UpdatePlayerESP()
    for _, p in pairs(Players:GetChildren()) do
        if p.Character and p.Character:FindFirstChild("Head") then
            local head = p.Character.Head
            if ESPPlayer and not head:FindFirstChild("NameEsp" .. Number) then
                local bg = Instance.new("BillboardGui", head)
                bg.Name = "NameEsp" .. Number
                bg.Size = UDim2.new(1, 200, 1, 30)
                bg.AlwaysOnTop = true
                bg.ExtentsOffset = Vector3.new(0, 1, 0)
                local txt = Instance.new("TextLabel", bg)
                txt.Size = UDim2.new(1, 0, 1, 0)
                txt.BackgroundTransparency = 1
                txt.TextStrokeTransparency = 0.5
                txt.Font = Enum.Font.GothamSemibold
                txt.TextSize = 14
                txt.TextColor3 = (p.Team == Player.Team)
                    and Color3.fromRGB(0, 255, 0) or Color3.fromRGB(255, 0, 0)
            elseif not ESPPlayer and head:FindFirstChild("NameEsp" .. Number) then
                head:FindFirstChild("NameEsp" .. Number):Destroy()
            end
            if ESPPlayer and head:FindFirstChild("NameEsp" .. Number) and p.Character:FindFirstChild("HumanoidRootPart") then
                local bg = head:FindFirstChild("NameEsp" .. Number)
                local d = round((Root.Position - head.Position).Magnitude / 3)
                local hp = round(p.Character.Humanoid.Health * 100 / p.Character.Humanoid.MaxHealth)
                bg.TextLabel.Text = p.Name .. " | " .. d .. " | HP " .. hp .. "%"
            end
        end
    end
end

task.spawn(function()
    while wait() do
        if ESPPlayer then pcall(UpdatePlayerESP) end
    end
end)

function Hop()
    local placeId = game.PlaceId
    local http    = game:GetService("HttpService")
    local cursor  = ""
    local seen    = {}
    while task.wait(1) do
        local url = "https://games.roblox.com/v1/games/" .. placeId ..
                    "/servers/Public?sortOrder=Asc&limit=100" ..
                    (cursor ~= "" and ("&cursor=" .. cursor) or "")
        local ok, resp = pcall(function()
            return http:JSONDecode(game:HttpGet(url))
        end)
        if not ok or not resp or not resp.data then return end
        cursor = (resp.nextPageCursor and resp.nextPageCursor ~= "null")
            and resp.nextPageCursor or ""
        for _, s in ipairs(resp.data) do
            if s.playing < s.maxPlayers and not seen[s.id] then
                seen[s.id] = true
                pcall(function()
                    game:GetService("TeleportService"):TeleportToPlaceInstance(placeId, s.id, Player)
                end)
                return
            end
        end
        if cursor == "" then break end
    end
end

Tabs.Misc:AddButton({ Title = "Server Hop", Callback = function() Hop() end })
