local STEP_SIZE, STEP_WAIT = 60, 0.5
function SafeMove(targetCF, checkStop)
    local hrp = Character and Character:FindFirstChild("HumanoidRootPart")
    if not hrp then return end
    local targetPos = targetCF.Position
    while true do
        if checkStop and checkStop() then break end
        local curPos = hrp.Position
        local rem = (targetPos - curPos).Magnitude
        if rem < 8 then break end
        local dir = (targetPos - curPos).Unit
        local step = math.min(rem, STEP_SIZE)
        local nextCF = (rem <= STEP_SIZE) and targetCF or CFrame.new(curPos + dir * step)
        pcall(function() hrp.CFrame = nextCF end)
        task.wait(STEP_WAIT)
    end
end

local _tweenThread = nil
function Tween(cf)
    if not cf then return end
    if _tweenThread then pcall(function() task.cancel(_tweenThread) end) end
    _tweenThread = task.spawn(function()
        SafeMove(cf, function() return _tweenThread == nil end)
        _tweenThread = nil
    end)
end
Tween2 = Tween

function AttackNoCoolDown()
    local tool = Character and Character:FindFirstChildOfClass("Tool")
    if not tool then return end
    local enemies = Workspace:FindFirstChild("Enemies")
    if not enemies then return end

    if tool:FindFirstChild("LeftClickRemote") then
        for _, mob in pairs(enemies:GetChildren()) do
            local hrp = mob:FindFirstChild("HumanoidRootPart")
            local hum = mob:FindFirstChild("Humanoid")
            if hrp and hum and hum.Health > 0 and (hrp.Position - Root.Position).Magnitude <= 60 then
                local dir = (hrp.Position - Root.Position).Unit
                pcall(function() tool.LeftClickRemote:FireServer(dir, 1) end)
            end
        end
    else
        local modNet = ReplicatedStorage:FindFirstChild("Modules") and ReplicatedStorage.Modules:FindFirstChild("Net")
        if not modNet then return end
        local regAtk = modNet:FindFirstChild("RE/RegisterAttack")
        local regHit = modNet:FindFirstChild("RE/RegisterHit")
        if not regAtk or not regHit then return end
        local hits, head = {}, nil
        for _, mob in pairs(enemies:GetChildren()) do
            local hum = mob:FindFirstChild("Humanoid")
            local hrp = mob:FindFirstChild("HumanoidRootPart")
            if hum and hrp and hum.Health > 0 and (hrp.Position - Root.Position).Magnitude <= 60 then
                local h = mob:FindFirstChild("Head")
                if h then hits[#hits+1] = {mob, h} head = head or h end
            end
        end
        if #hits > 0 then
            pcall(function()
                regAtk:FireServer(1e-9)
                regHit:FireServer(head, hits)
            end)
        end
    end
end

function Hop()
    local http = game:GetService("HttpService")
    local cursor, seen = "", {}
    while task.wait(1) do
        local url = "https://games.roblox.com/v1/games/"..game.PlaceId.."/servers/Public?sortOrder=Asc&limit=100"..(cursor~="" and ("&cursor="..cursor) or "")
        local ok, r = pcall(function() return http:JSONDecode(game:HttpGet(url)) end)
        if not ok or not r or not r.data then return end
        cursor = (r.nextPageCursor and r.nextPageCursor~="null") and r.nextPageCursor or ""
        for _, s in ipairs(r.data) do
            if s.playing < s.maxPlayers and not seen[s.id] then
                seen[s.id] = true
                pcall(function()
                    game:GetService("TeleportService"):TeleportToPlaceInstance(game.PlaceId, s.id, Player)
                end)
                return
            end
        end
        if cursor == "" then break end
    end
end

TabMisc:CreateButton("BtnHop", { Name = "Server Hop", Callback = function() Hop() end })
