-- 2. BỘ LỌC TỰ ĐỘNG THEO TEAM ID CỦA GAME
-- ==========================================

local function GetValidTargets()
    local targets = {}
    local myChar = LocalPlayer.Character
    if not myChar then return targets end

    -- Lấy thông tin Team của bạn trong trận đấu hiện tại
    local myTeamId = LocalPlayer:GetAttribute("MatchTeam")
    local myPlayerObj = Players:FindFirstChild(LocalPlayer.Name)

    local function IsAlive(obj)
        local hum = obj:FindFirstChildOfClass("Humanoid")
        if hum and hum.Health <= 0 then return false end
        local bHp = obj:GetAttribute("BattleHP")
        if bHp and type(bHp) == "number" and bHp <= 0 then return false end
        return true
    end

    for _, obj in ipairs(Workspace:GetChildren()) do
        if obj:IsA("Model") and obj.Name ~= LocalPlayer.Name and obj ~= myChar then
            local hrp = obj:FindFirstChild("HumanoidRootPart")
            if hrp and IsAlive(obj) then
                local targetName = obj.Name
                local isTeammate = false

                -- Kiểm tra nếu trùng tên chính mình
                if targetName == LocalPlayer.Name then
                    isTeammate = true
                else
                    -- Tự động quét đối chiếu Team ID (Attribute MatchTeam trên Player hoặc Model)
                    local targetPlayer = Players:FindFirstChild(targetName)
                    if targetPlayer then
                        local targetTeamId = targetPlayer:GetAttribute("MatchTeam")
                        -- Nếu có chung MatchTeam và khác nil thì tính là đồng đội
                        if myTeamId and targetTeamId and myTeamId == targetTeamId then
                            isTeammate = true
                        end
                    end
                    
                    -- Kiểm tra dự phòng qua thuộc tính Team của Roblox nếu game có dùng tính năng Team mặc định
                    if myPlayerObj and targetPlayer then
                        if myPlayerObj.Team and targetPlayer.Team and myPlayerObj.Team == targetPlayer.Team then
                            isTeammate = true
                        end
                    end
                end

                -- Chỉ nhắm vào những ai KHÔNG PHẢI là đồng đội trong trận
                if not isTeammate then
                    table.insert(targets, obj)
                end
            end
        end
    end

    return targets
end

-- [TÍNH NĂNG 1]: ESP KẺ ĐỊCH
local function ClearESP()
    for _, obj in ipairs(Workspace:GetDescendants()) do
        if obj.Name == "CyberESP_Highlight" or obj.Name == "CyberESP_Tag" then
            obj:Destroy()
        end
    end
end

local function UpdateESP()
    if not State.ESP then return end
    
    for _, obj in ipairs(Workspace:GetDescendants()) do
        if obj.Name == "CyberESP_Highlight" then
            local targetModel = obj.Adornee
            local hum = targetModel and targetModel:FindFirstChildOfClass("Humanoid")
            local bHp = targetModel and targetModel:GetAttribute("BattleHP")
            if not targetModel or not targetModel.Parent or (hum and hum.Health <= 0) or (bHp and bHp <= 0) then
                obj:Destroy()
                local hrp = targetModel and targetModel:FindFirstChild("HumanoidRootPart")
                if hrp and hrp:FindFirstChild("CyberESP_Tag") then
                    hrp.CyberESP_Tag:Destroy()
                end
            end
        end
    end

    local targets = GetValidTargets()
    for _, target in ipairs(targets) do
        if not target:FindFirstChild("CyberESP_Highlight") then
            local hl = Instance.new("Highlight")
            hl.Name = "CyberESP_Highlight"
            hl.Adornee = target
            hl.FillColor = Theme.Danger
            hl.FillTransparency = 0.65
            hl.OutlineColor = Theme.Accent
            hl.OutlineTransparency = 0
            hl.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
            hl.Parent = target
        end

        local hrp = target:FindFirstChild("HumanoidRootPart")
        if hrp and not hrp:FindFirstChild("CyberESP_Tag") then
            local bb = Instance.new("BillboardGui")
            bb.Name = "CyberESP_Tag"
            bb.Adornee = hrp
            bb.Size = UDim2.new(0, 120, 0, 30)
            bb.StudsOffset = Vector3.new(0, 3.2, 0)
            bb.AlwaysOnTop = true
            bb.Parent = hrp

            local txt = Instance.new("TextLabel")
            txt.Name = "Info"
            txt.Size = UDim2.new(1, 0, 1, 0)
            txt.BackgroundTransparency = 1
            txt.TextColor3 = Theme.Accent
            txt.TextStrokeTransparency = 0.3
            txt.Font = Enum.Font.Code
            txt.TextSize = 12
            txt.Parent = bb
        end

        if hrp and hrp:FindFirstChild("CyberESP_Tag") then
            local hp = target:GetAttribute("BattleHP") or (target:FindFirstChildOfClass("Humanoid") and math.floor(target:FindFirstChildOfClass("Humanoid").Health)) or "?"
            hrp.CyberESP_Tag.Info.Text = string.format("[ENEMY]\nHP: %s", tostring(hp))
        end
    end
end

-- [TÍNH NĂNG 2]: AIM-LOCK 360 AN TOÀN TUYỆT ĐỐI
local function GetClosestTarget360()
    local myChar = LocalPlayer.Character
    local myHrp = myChar and myChar:FindFirstChild("HumanoidRootPart")
    if not myHrp then return nil end

    local closestTarget = nil
    local shortestDist = math.huge

    for _, target in ipairs(GetValidTargets()) do
        local hrp = target:FindFirstChild("HumanoidRootPart")
        if hrp then
            local dist = (myHrp.Position - hrp.Position).Magnitude
            if dist < shortestDist then
                shortestDist = dist
                closestTarget = hrp
            end
        end
    end

    return closestTarget
end

local function AttemptServerHeal()
    local char = LocalPlayer.Character
    if char then
        local maxHp = char:GetAttribute("BattleMaxHP") or 100
        pcall(function() char:SetAttribute("BattleHP", maxHp) end)
        local hum = char:FindFirstChildOfClass("Humanoid")
        if hum then hum.Health = hum.MaxHealth end
    end
    SetLog("LOCAL HP SYNCED!", false)
end

            hum.MaxHealth = math.huge
            hum.Health = math.huge
        end
        SetLog("GOD MODE: ACTIVE", false)

        if hum then
            hum.MaxHealth = 100
            hum.Health = math.min(hum.Health, 100)
        end
        SetLog("GOD MODE: OFF", false)
    end
end)

RunService.Heartbeat:Connect(function()
    if State.GodMode then
        local char = LocalPlayer.Character
        if char then
            local hum = char:FindFirstChildOfClass("Humanoid")
            if hum then
                if hum.MaxHealth ~= math.huge then
                    hum.MaxHealth = math.huge
                end
                if hum.Health < math.huge then
                    hum.Health = math.huge
                end
            end
            local maxHpAttr = char:GetAttribute("BattleMaxHP")
            if maxHpAttr then
                char:SetAttribute("BattleHP", 999999)
            end
        end
    end
end)

RunService.RenderStepped:Connect(function()
    if State.ESP then UpdateESP() end
    if State.Aim360 then
        local targetPart = GetClosestTarget360()
        if targetPart then
            local camPos = Camera.CFrame.Position
            Camera.CFrame = CFrame.new(camPos, targetPart.Position)

            local myChar = LocalPlayer.Character
            local myHrp = myChar and myChar:FindFirstChild("HumanoidRootPart")
            if myHrp then
                local lookPos = Vector3.new(targetPart.Position.X, myHrp.Position.Y, targetPart.Position.Z)
                myHrp.CFrame = CFrame.new(myHrp.Position, lookPos)
            end
        end
    end
end)
